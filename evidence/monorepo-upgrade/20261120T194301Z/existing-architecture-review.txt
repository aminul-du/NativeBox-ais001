0005: 
0006: ## Stage 0 — Preserve and inspect

0008: - List archive contents before extraction.
0009: - Reject absolute paths, parent traversal and symlinks escaping extraction root.
0010: - Extract into a new review directory; never overwrite existing files.
0011: - Compare archive contents with repository paths and generate a missing-file report.
0012: 
0013: ## Stage 1 — Static audit
0014: - Run scripts/audit-repo.sh and scripts/validate-compose.sh .env.example.
0015: - Inspect Dockerfiles, dependency lockfiles, container users, ports, volumes and health checks.
0016: - Verify PostgreSQL and Qdrant are not published on host interfaces.
0017: [REDACTED: sensitive configuration line]
0018: 
0019: ## Stage 2 — Local build and tests
0020: - Build images without starting production services.
0021: - Run tests in isolation.
0022: - Verify Python/Node versions from manifests and lockfiles.
0023: - Verify Playwright browser dependencies and sandbox isolation.
0024: - Record commit SHA, image digests and test results.
0025: 
0026: ## Stage 3 — Staging
0027: [REDACTED: sensitive configuration line]
0028: - Require explicit approval before starting containers or changing network state.
0029: - Keep databases private and production hostname unchanged.
0030: [REDACTED: sensitive configuration line]
0031: 
0032: ## Stage 4 — Production approval
0033: AIS must explicitly approve the exact release commit, affected services, hostname, rollback plan and cost implications. Approval for one release does not authorize future releases or paid provisioning.
0034: 
0035: ## Stage 5 — Production verification
0036: - Verify tunnel health and hostname routing.
0037: - Verify application health externally.
0038: - Verify access controls and no unintended database exposure.

0001: #!/usr/bin/env bash
0002: set -Eeuo pipefail
0003: ARCHIVE="${1:-}"
0004: [[ -n "$ARCHIVE" && -f "$ARCHIVE" ]] || {
0005:   echo "Usage: bash scripts/audit-zip.sh /path/to/mcore-architecture.zip" >&2
0006:   exit 2
0007: }
0008: command -v unzip >/dev/null 2>&1 || { echo "unzip is required." >&2; exit 2; }
0009: command -v sha256sum >/dev/null 2>&1 || { echo "sha256sum is required." >&2; exit 2; }
0010: echo "Archive: $ARCHIVE"
0011: sha256sum "$ARCHIVE"
0012: tmp="$(mktemp)"
0013: trap 'rm -f "$tmp"' EXIT
0014: unzip -Z1 "$ARCHIVE" > "$tmp"
0015: printf '\nEntry count: '
0016: wc -l < "$tmp"
0017: unsafe=0
0018: while IFS= read -r path; do
0019:   case "$path" in
0020:     /*|../*|*/../*|*/..)
0021:       printf '[UNSAFE PATH] %s\n' "$path"; unsafe=1 ;;
0022:   esac
0023:   case "$path" in
0024:     .env|*/.env|.env.*|*/.env.*|*.pem|*.key|*/id_rsa|*/id_ed25519)
0025: [REDACTED: sensitive configuration line]
0026:   esac
0027: done < "$tmp"
0028: echo
0029: echo "Top-level paths:"
0030: awk -F/ 'NF { print $1 }' "$tmp" | sort -u
0031: if (( unsafe )); then
0032:   echo "REJECT: unsafe archive paths found. Do not extract this archive." >&2
0033:   exit 3
0034: fi
